// Learned baseline adjustment offsets (updated when user clicks "Learn Baseline")
var learnedOffset = { vib: 0, temp: 0 };

// Expected healthy vibration RMS (mm/s) as a smooth function of RPM and motor load current (A)
function expectedHealthyRms(rpm, current){
  var rpmNorm = clamp((rpm - 600) / 1200, 0, 1.5);
  var loadNorm = clamp((current - 7.5) / 17.5, 0, 1.5);
  return Math.max(0.6, 0.95 + 1.15 * rpmNorm + 0.85 * Math.pow(loadNorm, 1.15) + learnedOffset.vib);
}
function expectedHealthyTemp(rpm, current){
  var rpmNorm = clamp((rpm - 600) / 1200, 0, 1.5);
  var loadNorm = clamp((current - 7.5) / 17.5, 0, 1.5);
  return 36.0 + 11.0 * rpmNorm + 16.5 * loadNorm + learnedOffset.temp;
}

// Comprehensive diagnostic & prognostic evaluation for any (rpm, current, vibRms, kurtosis, temp, dutyHours)
function evaluateMotorCondition(d){
  var rpm = Number(d.rpm) || 1480;
  var cur = Number(d.current) || 14.8;
  var vib = Number(d.vibRms) || 2.1;
  var kurt = Number(d.kurtosis) || 3.0;
  var temp = Number(d.temp) || 52.0;
  var duty = clamp(Number(d.duty) || 16, 1, 24);

  var baseRms = expectedHealthyRms(rpm, cur);
  var baseTemp = expectedHealthyTemp(rpm, cur);
  var deltaVib = vib - baseRms;
  var deltaTemp = temp - baseTemp;
  var sigma = 0.28 + 0.08 * baseRms;
  var zScore = Math.max(0, deltaVib / sigma);

  // Impulsive bearing fault penalty from kurtosis > 3.4
  var kurtExcess = Math.max(0, kurt - 3.35);
  var tempExcess = Math.max(0, deltaTemp - 5.0);

  // Composite severity index (0 = pristine, 100 = imminent failure)
  var severity = clamp(
    Math.max(0, deltaVib) * 14.5 +
    kurtExcess * 9.5 +
    tempExcess * 1.45,
    0, 100
  );
  var health = clamp(100 - severity, 2, 100);

  var status = 'NORMAL';
  if (health < 55 || deltaVib >= 3.4 || kurt >= 6.2 || temp >= 88) status = 'CRITICAL';
  else if (health < 82 || deltaVib >= 1.35 || kurt >= 4.2 || temp >= 74) status = 'WARNING';

  // Likely root cause classification
  var faultType = 'Healthy Operating Envelope';
  var reqs = [];
  var summary = '';

  if (status === 'NORMAL'){
    faultType = 'Healthy Baseline Matched';
    summary = 'Vibration (' + vib.toFixed(2) + ' mm/s) and thermal profile (' + temp.toFixed(1) + ' °C) align with the learned healthy baseline (' + baseRms.toFixed(2) + ' mm/s) for ' + rpm + ' RPM and ' + cur.toFixed(1) + ' A load. No false alarm triggered.';
    reqs = [
      '<strong>Routine Lubrication:</strong> Maintain standard lithium-complex grease schedule every 2,000 operating hours.',
      '<strong>Baseline Tracking:</strong> Raspberry Pi edge daemon continues passive drift logging; no immediate parts required.',
      '<strong>Next Inspection:</strong> Standard visual coupling and terminal box check during planned turnaround.'
    ];
  } else if (kurt >= 4.3 && kurtExcess * 8 >= Math.max(0, deltaVib) * 10){
    faultType = 'Rolling-Element Bearing Wear (High Kurtosis)';
    summary = 'Elevated kurtosis (' + kurt.toFixed(2) + ') with high-frequency spectral energy indicates early-to-moderate inner/outer race spalling on the drive-end bearing.';
    reqs = [
      '<strong>Replacement Parts:</strong> Drive-End Deep Groove Ball Bearing (SKF 6309-2Z / C3 clearance) &amp; shaft lip seal.',
      '<strong>Lubrication Action:</strong> Purge old grease and replenish with high-load synthetic polyurea bearing grease.',
      '<strong>Maintenance Window:</strong> ' + (status === 'CRITICAL' ? 'Immediate shutdown within 24–48 hours to prevent rotor-stator rub.' : 'Schedule bearing swap within next 7–14 days.')
    ];
  } else if (tempExcess > 10 && deltaVib < 2.2){
    faultType = 'Thermal Overload / Cooling Restriction';
    summary = 'Casing temperature (' + temp.toFixed(1) + ' °C) exceeds the learned thermal curve by +' + deltaTemp.toFixed(1) + ' °C despite moderate mechanical vibration.';
    reqs = [
      '<strong>Cooling Inspection:</strong> Clear dust/debris from motor cooling fins, fan cowl, and gearbox breather vent.',
      '<strong>Electrical Check:</strong> Verify 3-phase current balance across Hall sensors and inspect terminal lugs for resistance heating.',
      '<strong>Lubrication Check:</strong> Check gearbox oil viscosity and level.'
    ];
  } else if (deltaVib >= 1.35){
    faultType = (rpm > 1350 && kurt < 4.0) ? 'Rotor Imbalance / Shaft Misalignment' : 'Mechanical Looseness / Coupling Wear';
    summary = 'Vibration exceeds the load-compensated baseline by +' + deltaVib.toFixed(2) + ' mm/s (' + zScore.toFixed(2) + ' σ), indicating rotational unbalance or angular coupling misalignment.';
    reqs = [
      '<strong>Mechanical Action:</strong> Perform single-plane dynamic rotor balancing and laser shaft-to-gearbox alignment.',
      '<strong>Hardware Required:</strong> Flexible spider coupling elastomer insert (NBR 98 Shore-A) &amp; stainless alignment shims.',
      '<strong>Fastener Torque:</strong> Re-torque motor mounting foot bolts and foundation anchor studs.'
    ];
  } else {
    faultType = 'Gradual Mechanical Drift';
    summary = 'Persistent low-level deviation (+' + deltaVib.toFixed(2) + ' mm/s) tracked over consecutive windows. Early intervention prevents unplanned downtime.';
    reqs = [
      '<strong>Condition Check:</strong> Inspect drive belt/coupling tension and verify ADXL1002 magnetic stud mount.',
      '<strong>Parts Standby:</strong> Stage spare bearing kit and coupling insert in maintenance crib.'
    ];
  }

  // Remaining Useful Life (RUL) estimation in operating hours
  var rulHours;
  if (status === 'NORMAL') rulHours = Math.round(1450 * (health / 100));
  else if (status === 'WARNING') rulHours = Math.round(180 + (health - 55) * 14);
  else rulHours = Math.max(12, Math.round(health * 2.4));

  var rulDays = Math.max(1, Math.round(rulHours / duty));
  var repairDate = new Date(Date.now() + rulDays * 86400000);
  var dateStr = repairDate.toLocaleDateString('en-US', { month:'short', day:'numeric', year:'numeric' });

  return {
    rpm: rpm, current: cur, vibRms: vib, kurtosis: kurt, temp: temp, duty: duty,
    baseRms: baseRms, baseTemp: baseTemp, deltaVib: deltaVib, zScore: zScore,
    health: health, status: status, faultType: faultType,
    summary: summary, reqs: reqs, rulHours: rulHours, rulDays: rulDays, repairDateStr: dateStr
  };
}

var _lv = new THREE.Vector3();
function updateLabels(){
  var W = stage.clientWidth, H = stage.clientHeight;
  for (var i = 0; i < labels.length; i++){
    var L = labels[i];
    var want = labelsOn && curLabelIds.indexOf(L.id) >= 0;
    var vis = false, sx = 0, sy = 0;
    if (want && ANCHOR[L.anchor]){
      _lv.copy(ANCHOR[L.anchor]).project(camera);
      sx = (_lv.x * 0.5 + 0.5) * W; sy = (-_lv.y * 0.5 + 0.5) * H;
      vis = _lv.z > -1 && _lv.z < 1 && sx > 0 && sx < W && sy > 0 && sy < H;
    }
    if (vis){
      var lw = L.el.offsetWidth || 160, lh = L.el.offsetHeight || 38;
      var left = clamp(sx + L.dx, 8, Math.max(8, W - lw - 8));
      var top  = clamp(sy + L.dy, 8, Math.max(8, H - lh - 8));
      L.el.style.transform = 'translate(' + left.toFixed(1) + 'px,' + top.toFixed(1) + 'px)';
      var ex = L.dx >= 0 ? left : left + lw, ey = top + lh / 2;
      L.ln.setAttribute('x1', sx.toFixed(1)); L.ln.setAttribute('y1', sy.toFixed(1));
      L.ln.setAttribute('x2', ex.toFixed(1)); L.ln.setAttribute('y2', ey.toFixed(1));
      L.dot.setAttribute('cx', sx.toFixed(1)); L.dot.setAttribute('cy', sy.toFixed(1));
    }
    if (vis !== L.on){
      L.on = vis;
      L.el.classList.toggle('on', vis); L.ln.classList.toggle('on', vis); L.dot.classList.toggle('on', vis);
    }
  }
}

  cur.ci = ci; cur.si = si; tour.t = 0;
  var to = shotState(sh); to.az = view.az + wrapPi(to.az - view.az);
  if (instant) { view = to; tween = null; }
  else tween = { from:copyView(view), to:to, t:0, dur:1.8 };
  if (changed){
    $('shotDots').innerHTML = '';
    ch.shots.forEach(function(){ $('shotDots').appendChild(document.createElement('i')); });
  }
  var dots = $('shotDots').children;
  for (var i = 0; i < dots.length; i++) dots[i].classList.toggle('on', i === si);
  chapBtns.forEach(function(b, k){ b.classList.toggle('active', k === ci); });
  curLabelIds = sh.labels || [];

  var capEl = $('caption');
  if (instant || !built){
    $('capStep').textContent = 'Chapter ' + (ci + 1) + ' of ' + CHAPTERS.length + ' · ' + ch.name;
    $('capTitle').textContent = sh.title; $('capText').textContent = sh.text; $('capView').textContent = 'Camera: ' + sh.view;
  } else {
    capEl.classList.add('swap');
    setTimeout(function(){
      $('capStep').textContent = 'Chapter ' + (ci + 1) + ' of ' + CHAPTERS.length + ' · ' + ch.name;
      $('capTitle').textContent = sh.title; $('capText').textContent = sh.text; $('capView').textContent = 'Camera: ' + sh.view;
      capEl.classList.remove('swap');
    }, 180);
  }
  built = true;
}
function goFlat(i){ i = (i + FLAT.length) % FLAT.length; goShot(FLAT[i].ci, FLAT[i].si); }

var feedEl = $('feedList');
function pushEdgeLog(status, msg){
  var li = document.createElement('li');
  li.className = status === 'CRITICAL' ? 'crit' : (status === 'WARNING' ? 'warn' : '');
  var now = new Date();
  var ts = now.toTimeString().slice(0,8);
  li.innerHTML = '<span class="dot"></span><span><b>[' + ts + ']</b> ' + msg + '</span>';

}

// FFT Spectrum arrays (48 frequency bins from 0 to 480 Hz)
var FFT_N = 48;
var fftLive = new Float32Array(FFT_N);
var fftBase = new Float32Array(FFT_N);

function stepSimulation(dt){
  sim.t += dt;
  sim.rpm += (sim.rpmTarget - sim.rpm) * Math.min(1, dt * 3.0);

  // Spin shaft & flywheel smoothly at proportional angular speed
  var omega = (sim.rpm / 60) * Math.PI * 2 * 0.14;
  rotorSpin.rotation.x = (rotorSpin.rotation.x - omega * dt) % (Math.PI * 2);

  // Compute physical sensor readings from Load, RPM, and injected Fault
  var loadFrac = sim.loadPct / 100;
  var rpmFrac = (sim.rpm - 600) / 1200;
  var targetCur = 7.5 + 16.5 * loadFrac + (sim.fault === 'misalign' ? 1.4 : 0) + (Math.random() - 0.5) * 0.12;
  sim.current += (targetCur - sim.current) * Math.min(1, dt * 4);

  var healthyRms = expectedHealthyRms(sim.rpm, sim.current);
  var faultVibAdd = 0, targetKurt = 3.02 + (Math.random() - 0.5) * 0.08, tempAdd = 0;

  if (sim.fault === 'imbalance'){
    faultVibAdd = 2.65 * (0.5 + 0.65 * rpmFrac);
    targetKurt = 3.35 + (Math.random() - 0.5) * 0.12;
    tempAdd = 6.5;
  } else if (sim.fault === 'bearing'){
    faultVibAdd = 2.15 + 0.8 * loadFrac;
    targetKurt = 6.85 + (Math.random() - 0.5) * 0.35;
    tempAdd = 14.5;
  } else if (sim.fault === 'misalign'){
    faultVibAdd = 2.35 * (0.6 + 0.5 * loadFrac);
    targetKurt = 3.95 + (Math.random() - 0.5) * 0.18;
    tempAdd = 9.0;
  }

  var targetVib = healthyRms + faultVibAdd + (Math.random() - 0.5) * 0.07;
  sim.vibRms += (targetVib - sim.vibRms) * Math.min(1, dt * 4);
  sim.kurtosis += (targetKurt - sim.kurtosis) * Math.min(1, dt * 4);

  var targetTemp = expectedHealthyTemp(sim.rpm, sim.current) + tempAdd;
  sim.temp += (targetTemp - sim.temp) * Math.min(1, dt * 0.8);

  // Evaluate context-aware state
  var ev = evaluateMotorCondition({
    rpm: Math.round(sim.rpm), current: sim.current, vibRms: sim.vibRms,
    kurtosis: sim.kurtosis, temp: sim.temp, duty: 16
  });
  sim.eval = ev;

  // Physical 3D motor micro-vibration & gearbox thermal emissive glow
  var shake = clamp((sim.vibRms - 1.2) * 0.0014, 0, 0.011);
  motorVibGroup.position.y = Math.sin(sim.t * 58) * shake;
  motorVibGroup.position.z = Math.cos(sim.t * 63) * shake * 0.7;
  gearMat.emissiveIntensity = clamp((sim.temp - 55) / 55, 0, 0.42);

  // Update 3D LEDs and Tower Beacon
  var stHex = ev.status === 'CRITICAL' ? 0xff4d4d : (ev.status === 'WARNING' ? 0xf2b84b : 0x00e5c7);
  accRing.material.color.setHex(stHex);
  accRing.scale.setScalar(0.16 + 0.05 * Math.sin(sim.t * 8));
  tower.green.color.setHex(ev.status === 'NORMAL' ? 0x22d36b : 0x0d2b1a);
  tower.amber.color.setHex(ev.status === 'WARNING' ? 0xf2b84b : 0x2e230d);
  tower.red.color.setHex(ev.status === 'CRITICAL' ? (Math.sin(sim.t * 14) > 0 ? 0xff4d4d : 0x661414) : 0x2b0d0d);

  // Log state transitions automatically
  if (ev.status !== sim.lastStatus){
    sim.lastStatus = ev.status;
    pushEdgeLog(ev.status, ev.status + ': ' + ev.faultType + ' (' + ev.vibRms.toFixed(2) + ' mm/s)');
  }
  sim.logT += dt;
  if (sim.logT >= 6.5){
    sim.logT = 0;
    pushEdgeLog(ev.status, 'Ctx ' + sim.loadPct + '% @ ' + Math.round(sim.rpm) + ' RPM → ' + ev.health.toFixed(0) + '% Health');
  }

  // Update FFT bins (1x shaft frequency = sim.rpm / 60 Hz; each bin = 10 Hz)
  var f1x = sim.rpm / 60;
  var bin1 = f1x / 10;
  var bin2 = bin1 * 2;
  for (var b = 0; b < FFT_N; b++){
    var baseNoise = 0.14 + 0.08 * Math.exp(-b / 18);
    var p1Base = (ev.baseRms * 0.72) * Math.exp(-Math.pow((b - bin1) / 1.1, 2));
    var p2Base = (ev.baseRms * 0.24) * Math.exp(-Math.pow((b - bin2) / 1.2, 2));
    fftBase[b] = baseNoise + p1Base + p2Base;

    var p1Live = p1Base * (sim.fault === 'imbalance' ? 3.2 : 1.0);
    var p2Live = p2Base * (sim.fault === 'misalign' ? 4.4 : 1.0);
    var hfLive = (sim.fault === 'bearing' && b > 24 && b < 42) ? (1.65 * Math.sin((b - 24) * 0.55) + Math.random() * 0.45) : 0;
    fftLive[b] = baseNoise + p1Live + p2Live + Math.max(0, hfLive) + Math.random() * 0.07;
  }

  // Push sparkline telemetry & refresh UI at 15 Hz
  sim.chartT += dt;
  if (sim.chartT >= 0.08){
    sim.chartT = 0;
    hist.vib.push(sim.vibRms); hist.cur.push(sim.current);
    hist.rpm.push(sim.rpm); hist.temp.push(sim.temp);
    for (var k in hist) if (hist[k].length > SPARK_N) hist[k].shift();
    updateTab1Hud();
  }

  sim.screenT += dt;
  if (sim.screenT >= 0.35){
    sim.screenT = 0;
    drawCabinetOled();
  }
}

function drawSpark(cv, arr, lo, hi, col){
  var g = cv.getContext('2d'), W = cv.width, H = cv.height;
  g.clearRect(0, 0, W, H);
  g.strokeStyle = col; g.lineWidth = 2; g.beginPath();
  for (var i = 0; i < arr.length; i++){
    var x = W - (arr.length - 1 - i) / (SPARK_N - 1) * W;
    var y = H - 3 - clamp((arr[i] - lo) / (hi - lo), 0, 1) * (H - 6);
    if (i) g.lineTo(x, y); else g.moveTo(x, y);
  }
  g.stroke();
}

function drawFftCanvas(){
  var cv = $('cvFft'), g = cv.getContext('2d'), W = cv.width, H = cv.height;
  g.fillStyle = '#071018'; g.fillRect(0, 0, W, H);

  // Grid lines
  g.strokeStyle = '#1B2E3A'; g.lineWidth = 1;
  for (var gy = 1; gy <= 4; gy++){
    var y = H - 24 - (gy / 4) * (H - 36);
    g.beginPath(); g.moveTo(34, y); g.lineTo(W - 10, y); g.stroke();
  }

  // Learned Healthy Envelope (dashed amber/cyan reference)
  g.save();
  g.setLineDash([5, 5]);
  g.strokeStyle = 'rgba(242,184,75,0.85)'; g.lineWidth = 2;
  g.beginPath();
  for (var i = 0; i < FFT_N; i++){
    var x = 34 + (i / (FFT_N - 1)) * (W - 44);
    var y = H - 24 - clamp(fftBase[i] / 5.5, 0, 1) * (H - 36);
    if (i) g.lineTo(x, y); else g.moveTo(x, y);
  }
  g.stroke();
  g.restore();

  // Live FFT Spectrum Curve
  var col = sim.eval && sim.eval.status === 'CRITICAL' ? '#FF4D4D' : (sim.eval && sim.eval.status === 'WARNING' ? '#F2B84B' : '#00E5C7');
  g.strokeStyle = col; g.lineWidth = 2.4;
  g.beginPath();
  for (var j = 0; j < FFT_N; j++){
    var x2 = 34 + (j / (FFT_N - 1)) * (W - 44);
    var y2 = H - 24 - clamp(fftLive[j] / 5.5, 0, 1) * (H - 36);
    if (j) g.lineTo(x2, y2); else g.moveTo(x2, y2);
  }
  g.stroke();

  g.fillStyle = '#83A0AA'; g.font = '12px "IBM Plex Mono", monospace';
  g.fillText('0 Hz', 34, H - 7);
  g.fillText('120 Hz', 34 + (W - 44)*0.25, H - 7);
  g.fillText('240 Hz (Bearing Band)', 34 + (W - 44)*0.5, H - 7);
  g.fillText('480 Hz', W - 54, H - 7);
}

function updateCamera(dt){
  if (tween){
    tween.t += dt;
    var k = easeIO(tween.t / tween.dur), f = tween.from, to = tween.to;
    view.tx = lerp(f.tx, to.tx, k); view.ty = lerp(f.ty, to.ty, k); view.tz = lerp(f.tz, to.tz, k);
    view.az = lerp(f.az, to.az, k); view.el = lerp(f.el, to.el, k); view.dist = lerp(f.dist, to.dist, k);
    if (tween.t >= tween.dur) tween = null;
  }
  var az = view.az + 0.03 * Math.sin(sim.t * 0.25);
  var el = clamp(view.el + 0.01 * Math.sin(sim.t * 0.3), deg(4), deg(82));
  var ce = Math.cos(el);
  camera.position.set(view.tx + view.dist * Math.sin(az) * ce, Math.max(0.18, view.ty + view.dist * Math.sin(el)), view.tz + view.dist * Math.cos(az) * ce);
  camera.lookAt(view.tx, view.ty, view.tz);
}

var STORE_KEY = 'motorsense_rpi_records_v1';

function defaultSeedRecords(){
  var now = Date.now();
  var seeds = [
    { ts: now - 86400000*6, rpm: 1480, current: 14.5, vibRms: 2.08, kurtosis: 3.01, temp: 51.5, duty: 16, note: 'Commissioning baseline capture' },
    { ts: now - 86400000*4, rpm: 1740, current: 21.8, vibRms: 2.78, kurtosis: 3.12, temp: 62.0, duty: 16, note: 'Peak production shift (high load, healthy)' },
    { ts: now - 86400000*2, rpm: 1480, current: 15.0, vibRms: 2.24, kurtosis: 3.18, temp: 53.2, duty: 16, note: 'Routine automated Raspberry Pi log' },
    { ts: now - 86400000*1, rpm: 1500, current: 15.4, vibRms: 2.48, kurtosis: 3.38, temp: 55.0, duty: 16, note: 'Slight 1x rotor drift observed' }
  ];
  return seeds.map(function(s){
    var ev = evaluateMotorCondition(s);
    ev.ts = s.ts; ev.note = s.note;
    return ev;
  });
}

var records = (function(){
  try {
    var raw = localStorage.getItem(STORE_KEY);
    if (raw){
      var parsed = JSON.parse(raw);
      if (Array.isArray(parsed) && parsed.length) return parsed;
    }
  } catch (e){}
  return defaultSeedRecords();
})();

function saveRecords(){
  try { localStorage.setItem(STORE_KEY, JSON.stringify(records)); } catch (e){}
  $('logCountBadge').textContent = records.length;
  $('tableMeta').textContent = records.length + ' record' + (records.length === 1 ? '' : 's') + ' stored in browser';
}

function addRecordFromInputs(data, note, silent){
  var ev = evaluateMotorCondition(data);
  ev.ts = Date.now();
  ev.note = note || 'Diagnostic entry';
  records.unshift(ev);
  if (records.length > 60) records.pop();
  saveRecords();
  renderTab2All(ev);
  if (!silent) toast('Analyzed & stored record (' + ev.status + ')');
}

function drawDegradationChart(activeEv){
  var cv = $('cvDegrade'), g = cv.getContext('2d'), W = cv.width, H = cv.height;
  g.fillStyle = '#071018'; g.fillRect(0, 0, W, H);

  var padL = 48, padR = 20, padT = 18, padB = 28;
  var pw = W - padL - padR, ph = H - padT - padB;
  var maxDev = 5.5; // mm/s deviation above baseline

  // Horizontal grid & Critical Threshold (4.5 mm/s)
  g.strokeStyle = '#1B2E3A'; g.lineWidth = 1;
  g.font = '11px "IBM Plex Mono", monospace'; g.fillStyle = '#83A0AA';
  [0, 1.5, 3.0, 4.5].forEach(function(val){
    var y = padT + ph - (val / maxDev) * ph;
    g.beginPath(); g.moveTo(padL, y); g.lineTo(W - padR, y); g.stroke();
    g.fillText('+' + val.toFixed(1), 6, y + 4);
  });

  // Critical threshold dashed red line at +4.5 mm/s
  var yCrit = padT + ph - (4.5 / maxDev) * ph;
  g.save(); g.setLineDash([6, 4]); g.strokeStyle = '#FF4D4D'; g.lineWidth = 1.6;
  g.beginPath(); g.moveTo(padL, yCrit); g.lineTo(W - padR, yCrit); g.stroke();
  g.restore();

  // Plot recent stored history on left 45% of chart, and future RUL trajectory on right 55%
  var histPts = records.slice(0, 8).reverse();
  if (activeEv && (!histPts.length || histPts[histPts.length - 1] !== activeEv)){
    histPts.push(activeEv);
  }
  var splitX = padL + pw * 0.44;

  // Vertical "NOW" marker
  g.strokeStyle = 'rgba(0,229,199,0.35)'; g.beginPath(); g.moveTo(splitX, padT); g.lineTo(splitX, padT + ph); g.stroke();
  g.fillStyle = '#00E5C7'; g.fillText('NOW', splitX - 12, H - 8);
  g.fillStyle = '#83A0AA'; g.fillText('HISTORICAL LOG', padL + 8, H - 8);
  g.fillText('PROJECTED DEGRADATION TO REPAIR WINDOW', splitX + 14, H - 8);

  if (histPts.length){
    g.strokeStyle = '#00E5C7'; g.lineWidth = 2.4; g.beginPath();
    histPts.forEach(function(pt, i){
      var x = histPts.length === 1 ? splitX : padL + (i / (histPts.length - 1)) * (splitX - padL);
      var dev = clamp(Math.max(0, pt.deltaVib), 0, maxDev);
      var y = padT + ph - (dev / maxDev) * ph;
      if (i) g.lineTo(x, y); else g.moveTo(x, y);
    });
    g.stroke();

    // Projected exponential degradation curve from NOW to RUL threshold
    var last = histPts[histPts.length - 1];
    var startDev = clamp(Math.max(0.08, last.deltaVib), 0.08, 4.4);
    var endX = clamp(splitX + (last.rulDays / 90) * (W - padR - splitX), splitX + 45, W - padR);

    g.save();
    g.setLineDash([5, 4]);
    g.strokeStyle = last.status === 'CRITICAL' ? '#FF4D4D' : '#F2B84B';
    g.lineWidth = 2.2;
    g.beginPath();
    for (var s = 0; s <= 24; s++){
      var t = s / 24;
      var px = lerp(splitX, endX, t);
      var pDev = startDev + (4.5 - startDev) * Math.pow(t, 1.65);
      var py = padT + ph - (clamp(pDev, 0, maxDev) / maxDev) * ph;
      if (s) g.lineTo(px, py); else g.moveTo(px, py);
    }
    g.stroke();
    g.restore();

    // Target repair marker dot
    g.fillStyle = '#F2B84B';
    g.beginPath(); g.arc(endX, yCrit, 5, 0, Math.PI * 2); g.fill();
    g.fillStyle = '#F2B84B';
    g.fillText('Repair: ' + last.repairDateStr, Math.min(endX - 45, W - 155), yCrit - 9);
  }
}

function renderLogTable(){
  var tb = $('logTbody'); tb.innerHTML = '';
  records.forEach(function(r, idx){
    var tr = document.createElement('tr');
    var dtStr = new Date(r.ts).toLocaleString([], { month:'short', day:'2-digit', hour:'2-digit', minute:'2-digit' });
    var stCol = r.status === 'CRITICAL' ? 'var(--red)' : (r.status === 'WARNING' ? 'var(--amber)' : 'var(--green)');
    tr.innerHTML =
      '<td>' + dtStr + '</td>' +
      '<td>' + r.rpm + '</td>' +
      '<td>' + r.current.toFixed(1) + ' A</td>' +
      '<td><b>' + r.vibRms.toFixed(2) + ' mm/s</b></td>' +
      '<td>' + r.baseRms.toFixed(2) + ' mm/s</td>' +
      '<td>' + r.kurtosis.toFixed(2) + '</td>' +
      '<td>' + r.temp.toFixed(1) + ' °C</td>' +
      '<td>' + r.health.toFixed(0) + '%</td>' +
      '<td style="color:' + stCol + ';font-weight:600;">' + r.status + '</td>' +
      '<td>' + r.repairDateStr + '</td>' +
      '<td><button class="del-btn" data-idx="' + idx + '">Delete</button></td>';
    tr.addEventListener('click', function(e){
      if (e.target.classList.contains('del-btn')) return;
      renderDiagnosisCard(r);
      drawDegradationChart(r);
    });
    tb.appendChild(tr);
  });

  tb.querySelectorAll('.del-btn').forEach(function(btn){
    btn.addEventListener('click', function(){
      var i = parseInt(this.dataset.idx, 10);
      records.splice(i, 1);
      saveRecords();
      renderTab2All();
    });
  });
}

  renderLogTable();
}
